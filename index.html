import http.server import socketserver import socket import os import urllib.parse import json import cgi PORT = 8080 UPLOAD_DIR = "shared_files" os.makedirs(UPLOAD_DIR, exist_ok=True) # 自分のローカルIPアドレスを取得する関数 def get_local_ip(): s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM) try: s.connect(("8.8.8.8", 80)) ip = s.getsockname()[0] except Exception: ip = "127.0.0.1" finally: s.close() return ip # ブラウザに表示するHTMLアプリ（UI） HTML_PAGE = """ Local File & Text Share

📱↔️💻 ローカル共有アプリ

📤 ファイル送信（アップロード）

📋 クリップボード・メモ共有

📁 共有フォルダ内のファイル一覧

    """ shared_clipboard_text = "" class CustomHandler(http.server.SimpleHTTPRequestHandler): def do_GET(self): global shared_clipboard_text if self.path == "/": self.send_response(200) self.send_header("Content-type", "text/html; charset=utf-8") self.end_headers() self.wfile.write(HTML_PAGE.encode("utf-8")) elif self.path == "/api/files": files = os.listdir(UPLOAD_DIR) self.send_response(200) self.send_header("Content-type", "application/json") self.end_headers() self.wfile.write(json.dumps(files).encode("utf-8")) elif self.path == "/api/text": self.send_response(200) self.send_header("Content-type", "application/json") self.end_headers() self.wfile.write(json.dumps({"text": shared_clipboard_text}).encode("utf-8")) elif self.path.startswith("/download/"): filename = urllib.parse.unquote(self.path[len("/download/"):]) filepath = os.path.join(UPLOAD_DIR, os.path.basename(filename)) if os.path.exists(filepath): self.send_response(200) self.send_header("Content-Disposition", f'attachment; filename="{filename}"') self.end_headers() with open(filepath, "rb") as f: self.wfile.write(f.read()) else: self.send_error(404) else: self.send_error(404) def do_POST(self): global shared_clipboard_text if self.path == "/api/upload": form = cgi.FieldStorage( fp=self.rfile, headers=self.headers, environ={'REQUEST_METHOD': 'POST', 'CONTENT_TYPE': self.headers['Content-Type']} ) file_items = form['files'] if isinstance(form['files'], list) else [form['files']] for item in file_items: if item.filename: filepath = os.path.join(UPLOAD_DIR, os.path.basename(item.filename)) with open(filepath, "wb") as f: f.write(item.file.read()) self.send_response(200) self.end_headers() elif self.path == "/api/text": content_length = int(self.headers['Content-Length']) post_data = self.rfile.read(content_length) data = json.loads(post_data.decode("utf-8")) shared_clipboard_text = data.get("text", "") self.send_response(200) self.end_headers() if __name__ == "__main__": local_ip = get_local_ip() print("=" * 50) print(f"サーバー起動中！同じWi-FiのAndroidブラウザで以下にアクセス：") print(f"👉 http://{local_ip}:{PORT}") print("=" * 50) with socketserver.ThreadingTCPServer(("", PORT), CustomHandler) as httpd: httpd.serve_forever()
